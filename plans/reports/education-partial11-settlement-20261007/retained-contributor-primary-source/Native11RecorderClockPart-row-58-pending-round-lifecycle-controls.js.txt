export async function observeQuizActive(root, { timed = false, optionId = 'release-single-choice/option-1' } = {}, evidence) {
  assertEvidenceSink(evidence)
  const sample = await readCorrelatedStateAndTime(root, evidence, 'observeQuizActive')
  const guard = await root.evaluate((node, optId) => {
    const submitBtn = node.querySelector('button[type="submit"]') || Array.from(node.querySelectorAll('button')).find(b => b.textContent?.trim() === 'Submit answer')
    const optInput = node.querySelector(`input[value="${optId}"]`)
    return {
      submitEnabled: Boolean(submitBtn && !submitBtn.disabled),
      optionEnabled: Boolean(optInput && !optInput.disabled),
    }
  }, optionId)
  evidence.stage('observeQuizActiveGuard', guard)
  await expect(root.getByRole('button', { name: 'Submit answer', exact: true })).toBeEnabled()
  await expect(root.locator(`input[value="${optionId}"]`)).toBeEnabled()
  const state = sample.state
  expect(state).not.toBeNull()
  expect(state.status).toBe('active')
  if (timed) expect(state.deadline).toBeGreaterThan(0)
  else expect(state.deadline).toBeNull()
  return state
}

export async function answerQuizQuestion(root, optionId = 'release-single-choice/option-1', evidence) {
  assertEvidenceSink(evidence)
  const option = root.locator(`input[value="${optionId}"]`)
  const submitBtn = root.getByRole('button', { name: 'Submit answer', exact: true })
  const checkBracket = await executeAction(root, () => option.check(), { type: 'change' }, evidence, 'quizOptionCheck')
  const submitBracket = await executeAction(root, () => submitBtn.click(), { command: 'submit', type: 'submit' }, evidence, 'quizAnswerSubmit')
  return { state: submitBracket.stateAfter, actionBracket: submitBracket, actionRecords: [checkBracket, submitBracket] }
}

export async function startHotPotatoRound(root, playerId = 'p1', name = 'Player 1', { timed = true } = {}, evidence) {
  assertEvidenceSink(evidence)
  const idInput = root.getByRole('textbox', { name: 'Participant ID', exact: true })
  const nameInput = root.getByRole('textbox', { name: 'Participant name', exact: true })
  const idBracket = await executeAction(root, () => idInput.fill(playerId), { type: 'input' }, evidence, 'hpFillId')
  const nameBracket = await executeAction(root, () => nameInput.fill(name), { type: 'input' }, evidence, 'hpFillName')
  const addParticipantBracket = await executeAction(root, () => root.locator('button[data-ns-command="add-participant"]').click(), { command: 'add-participant', type: 'submit' }, evidence, 'hpAddParticipantAction')
  const actionBracket = await executeAction(root, () => root.locator('button[data-ns-command="start"]').click(), { command: 'start', type: 'click' }, evidence, 'hpStartAction')
  const state = actionBracket.stateAfter
  expect(state).not.toBeNull()
  expect(state.phase).toBe('active')
  if (timed) expect(state.deadline).not.toBeNull()
  else expect(state.deadline).toBeNull()
  return {
    state, actionBracket, addParticipantBracket,
    actionRecords: [idBracket, nameBracket, addParticipantBracket, actionBracket],
    startBracket: [actionBracket.tCapture, actionBracket.tBubble],
  }
}
