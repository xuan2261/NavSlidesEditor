function deepSnapshot(value) {
  if (value === null || typeof value !== 'object') return value
  if (value instanceof Error) {
    return {
      name: value.name,
      message: value.message,
      stack: value.stack || null,
      cause: value.cause ? deepSnapshot(value.cause) : null,
    }
  }
  if (Array.isArray(value)) return value.map(deepSnapshot)
  const copy = {}
  for (const [k, v] of Object.entries(value)) copy[k] = deepSnapshot(v)
  return copy
}
