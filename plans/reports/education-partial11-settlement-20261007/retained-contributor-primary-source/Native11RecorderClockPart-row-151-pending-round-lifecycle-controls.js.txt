}

export async function startTriviaRound(root, evidence) {
  assertEvidenceSink(evidence)
  const startBtn = root.locator('button[data-ns-command="start"]')
  const actionBracket = await executeAction(root, () => startBtn.click(), { command: 'start', type: 'click' }, evidence, 'triviaStartAction')
  const state = actionBracket.stateAfter
  expect(state).not.toBeNull()
  expect(state.phase).toBe('active')
  expect(state.deadline).not.toBeNull()
  return { state, actionBracket, actionRecords: [actionBracket], startBracket: [actionBracket.tCapture, actionBracket.tBubble] }
}

export async function answerTriviaOption(root, optionId = 'question-1-option-1', evidence) {

export async function resetGame(root, evidence) {
  assertEvidenceSink(evidence)
  const btn = root.locator('button[data-ns-command="reset"], button[data-ns-command="quiz-reset"]')
  const guard = await stageButtonGuard(root, 'button[data-ns-command="reset"], button[data-ns-command="quiz-reset"]', 'resetGameGuard', evidence)
  await expect(btn).toBeVisible()
  const cmd = guard.buttonCommand
  const actionBracket = await executeAction(root, () => btn.click(), { command: cmd, type: 'click' }, evidence, 'resetGameAction')
  return { state: actionBracket.stateAfter, actionBracket, actionRecords: [actionBracket] }
}
