import { expect } from '@playwright/test'
import { apiGetPresentation } from '../fixtures/test-fixtures.js'
import { collectLivePageDiagnostics, captureLiveScreenshot } from './pending-round-lifecycle-live-evidence.js'
import { createErrorLedger } from './pending-round-lifecycle-error-ledger.js'
import { validateManifest, validateDocumentIdentity } from './pending-round-lifecycle-evidence-domain.js'

function deepSnapshot(value) {
  if (value === null || typeof value !== 'object') return value
  if (value instanceof Error) {
    return {
      name: value.name,
      message: value.message,
      stack: value.stack || null,
      cause: value.cause ? deepSnapshot(value.cause) : null,
    }
  }
  if (Array.isArray(value)) return value.map(deepSnapshot)
  const copy = {}
  for (const [k, v] of Object.entries(value)) copy[k] = deepSnapshot(v)
  return copy
}

export function createEvidenceAccumulator(testInfo, testName) {
  const records = {}
  const stageTrace = []
  let activePage = null
  let activePageMetadata = null
  let expectedRoots = null
  let boundDocumentIdentity = null
  let plannedUrl = null
  const ledger = createErrorLedger()

  const sink = {
    stage(label, payload) {
      if (!label || typeof label !== 'string') {
        throw new Error('stage requires a non-empty string label')
      }
      const snap = deepSnapshot(payload)
      const entry = {
        label,
        payload: snap,
        time: performance.now(),
        order: stageTrace.length + 1,
      }
      stageTrace.push(entry)
      records[label] = snap
    },

    setActivePage(page, metadata = null) {
      if (!metadata || typeof metadata !== 'object' || !metadata.surface || !metadata.plannedUrl) {
        throw new Error('setActivePage requires valid metadata with surface and plannedUrl')
      }
      activePage = page
      activePageMetadata = metadata
      plannedUrl = metadata.plannedUrl
      if (metadata.expectedRoots) {
        sink.setExpectedRoots(metadata.expectedRoots)
      }
      sink.stage('activePageMetadata', metadata)
      if (page && typeof page.on === 'function') {
        page.on('pageerror', (err) => ledger.recordError('page:error', err))
      }
    },

    setExpectedRoots(manifest) {
      validateManifest(manifest)
      expectedRoots = manifest
      sink.stage('expectedRootsManifest', manifest)
    },

    getExpectedRoots() {
      return expectedRoots
    },

    bindDocumentIdentity(snapshot) {
      if (boundDocumentIdentity) {
        throw new Error('Document identity already bound; cannot re-bind journey identity')
      }
      validateDocumentIdentity(snapshot, plannedUrl, null)
      boundDocumentIdentity = deepSnapshot(snapshot)
      sink.stage('documentIdentityBound', boundDocumentIdentity)
      return boundDocumentIdentity
    },

    assertDocumentIdentity(snapshot) {
      if (!boundDocumentIdentity) {
        throw new Error('Document identity has not been bound; bindDocumentIdentity is required first')
      }
      validateDocumentIdentity(snapshot, plannedUrl, boundDocumentIdentity)
      return snapshot
    },

    getPrimaryError() {
      return ledger.getPrimaryError()
    },

    produceThrow() {
      return ledger.produceThrow()
    },

    recordError(phase, err, metadata = null) {
      if (!phase || typeof phase !== 'string') {
        throw new Error('recordError requires a non-empty string phase')
      }
      return ledger.recordError(phase, err, metadata)
    },

    recordSecondaryError(err) {
      ledger.recordError('secondary', err)
    },

    async collectLive(overridePage = null) {
      const pageToInspect = overridePage || activePage
      if (!pageToInspect || pageToInspect.isClosed()) {
        ledger.recordError('collectLive', new Error('Live collection failed: page is missing or closed'))
        return
      }

      try {
        const liveDiagnostics = await collectLivePageDiagnostics(pageToInspect, expectedRoots, sink)
        records.liveDiagnostics = liveDiagnostics
      } catch (diagError) {
        ledger.recordError('collectLiveDiagnostics', diagError)
      }

      try {
        const screenshot = await captureLiveScreenshot(pageToInspect)
        if (screenshot) {
          await testInfo.attach(`${testName}-shot`, {
            body: screenshot,
            contentType: 'image/png',
          })
        }
      } catch (shotError) {
        ledger.recordError('collectLiveScreenshot', shotError)
      }
    },

    async finalize(request, presentationId, savedBeforeJSON, runnerError = null) {
      if (runnerError) {
        ledger.recordError('runner', runnerError)
      }

      if (!savedBeforeJSON) {
        records.sourceInvariantVerified = false
        records.sourceInvariantError = 'Baseline savedBeforeJSON missing'
        ledger.recordError('sourceInvariant:prerequisite', new Error('Baseline savedBeforeJSON is missing; explicit prerequisite failure'))
      }

      let savedAfter = null
      try {
        savedAfter = await apiGetPresentation(request, presentationId)
        records.savedAfter = savedAfter
        if (savedBeforeJSON) {
          expect(savedAfter).toEqual(savedBeforeJSON)
          records.sourceInvariantVerified = true
        }
      } catch (getOrMatchError) {
        records.sourceInvariantVerified = false
        records.sourceInvariantError = getOrMatchError?.message || String(getOrMatchError)
        ledger.recordError('sourceInvariant', getOrMatchError)
      }

      records.stageTrace = stageTrace
      records.errors = ledger.getSerializedErrors()
      const primaryErr = ledger.getPrimaryError()
      if (primaryErr) {
        records.primaryError = {
          name: primaryErr.name,
          message: primaryErr.message || String(primaryErr),
          stack: primaryErr.stack || null,
        }
        if (primaryErr.rootInstanceId) {
          records.primaryError.rootInstanceId = primaryErr.rootInstanceId
        }
        if (primaryErr.timeOrigin !== undefined && primaryErr.timeOrigin !== null) {
          records.primaryError.timeOrigin = primaryErr.timeOrigin
        }
      }

      try {
        await testInfo.attach(`${testName}-measured`, {
          body: JSON.stringify(records, null, 2),
          contentType: 'application/json',
        })
      } catch (attachError) {
        ledger.recordError('attachment', attachError)
      }

      const toThrow = ledger.produceThrow()
      if (toThrow) {
        throw toThrow
      }
    },
  }

  return sink
}
