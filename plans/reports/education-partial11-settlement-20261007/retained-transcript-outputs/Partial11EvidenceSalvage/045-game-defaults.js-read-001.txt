function sampleQuestion(id = 'question-1', count = 2, points = 10) {
  const options = Array.from({ length: count }, (_, i) => ({
    id: `${id}-option-${i + 1}`,
    text: i === 0 ? 'Paris' : ['Rome', 'London', 'Berlin'][i - 1],
  }))
  return {
    id,
    kind: 'single-choice',
    prompt: 'What is the capital of France?',
    points,
    timeLimit: 30,
    options,
    answerKey: { optionId: options[0].id },
    feedback: {
      correct: 'Correct!',
      incorrect: 'Try again.',
      explanation: 'Paris is the capital of France.',
    },
  }
}

}
