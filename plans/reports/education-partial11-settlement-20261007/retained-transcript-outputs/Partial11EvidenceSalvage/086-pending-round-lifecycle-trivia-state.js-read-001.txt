import { expect } from '@playwright/test'

export const TRIVIA_KEYS = [
  'phase',
  'error',
  'roundIndex',
  'roundId',
  'questionIndex',
  'questionId',
  'turn',
  'scores',
  'awards',
  'buzzTeamId',
  'bonus',
  'roundDeadline',
  'questionDeadline',
  'startedAt',
  'deadline',
  'roundExpired',
  'correct',
  'winnerId',
  'generation',
  'runtimeMode',
]

export function assertTriviaKeys(state) {
  expect(Object.keys(state).sort()).toEqual([...TRIVIA_KEYS].sort())
}

export function assertTriviaInitialOrReset(state, expectedConfig, generation = 0) {
  assertTriviaKeys(state)
  expect(state.runtimeMode).toBe('local')
  expect(state.generation).toBe(generation)
  expect(state.phase).toBe('setup')
  expect(state.error).toBeNull()
  expect(state.roundIndex).toBe(-1)
  expect(state.roundId).toBeNull()
  expect(state.questionIndex).toBe(-1)
  expect(state.questionId).toBeNull()
  expect(state.turn).toBe(0)
  expect(state.scores).toEqual(Object.fromEntries(expectedConfig.teams.map((t) => [t.id, 0])))
  expect(state.awards).toEqual([])
  expect(state.buzzTeamId).toBeNull()
  expect(state.bonus).toBe('normal')
  expect(state.roundDeadline).toBeNull()
  expect(state.questionDeadline).toBeNull()
  expect(state.startedAt).toBeNull()
  expect(state.deadline).toBeNull()
  expect(state.roundExpired).toBe(false)
  expect(state.correct).toBeNull()
  expect(state.winnerId).toBeNull()
}

export function assertTriviaActive(state, expectedConfig, generation = 0) {
  assertTriviaKeys(state)
  expect(state.runtimeMode).toBe('local')
  expect(state.generation).toBe(generation)
  expect(state.phase).toBe('active')
  expect(state.error).toBeNull()
  expect(state.roundIndex).toBe(0)
  expect(state.roundId).toBe('round-1')
  expect(state.questionIndex).toBe(0)
  expect(state.questionId).toBe('q1')
  expect(state.turn).toBe(1)
  expect(state.scores).toEqual(Object.fromEntries(expectedConfig.teams.map((t) => [t.id, 0])))
  expect(state.awards).toEqual([])
  expect(state.buzzTeamId).toBeNull()
  expect(state.bonus).toBe('normal')
  expect(state.correct).toBeNull()
  expect(state.roundExpired).toBe(false)
  expect(state.winnerId).toBeNull()
  expect(typeof state.startedAt).toBe('number')
  expect(typeof state.roundDeadline).toBe('number')
  expect(typeof state.questionDeadline).toBe('number')
  expect(state.deadline).toBe(state.questionDeadline)
}

export function assertTriviaPeerSettled(state) {
  assertTriviaKeys(state)
  expect(state.runtimeMode).toBe('local')
  expect(state.generation).toBe(0)
  expect(state.phase).toBe('review')
  expect(state.error).toBeNull()
  expect(state.roundIndex).toBe(0)
  expect(state.roundId).toBe('round-1')
  expect(state.questionIndex).toBe(0)
  expect(state.questionId).toBe('q1')
  expect(state.turn).toBe(1)
  expect(state.scores).toEqual({ 'team-1': 0, 'team-2': 20 })
  const expectedAwards = ['0:trivia-champ:["round-1","q1",1]:buzz', '0:trivia-champ:["round-1","q1",1]:answer']
  expect(state.awards).toEqual(expectedAwards)
  expect(state.buzzTeamId).toBe('team-2')
  expect(state.bonus).toBe('normal')
  expect(state.roundExpired).toBe(false)
  expect(state.correct).toBe(true)
  expect(state.winnerId).toBeNull()
}

export async function assertTriviaPeerUI(root) {
  const scores = root.locator('ul[aria-label="Scores"] > li')
  await expect(scores.filter({ hasText: 'Red: 20' })).toBeVisible()
  await expect(scores.filter({ hasText: 'Blue: 0' })).toBeVisible()
}

export async function assertTriviaResetUI(root) {
  const scores = root.locator('ul[aria-label="Scores"] > li')
  await expect(scores.filter({ hasText: 'Blue: 0' })).toBeVisible()
  await expect(scores.filter({ hasText: 'Red: 0' })).toBeVisible()
}
