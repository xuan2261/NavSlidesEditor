import { gameFixture, deckWith } from './release-game-fixtures.js'

export function makeTriviaFixture(roundLimit, q1Limit, q2Limit = null, elementId = 'g-pri') {
  const g = gameFixture('trivia-champ')
  g.id = elementId
  g['trivia-champ'].rounds[0].timeLimit = roundLimit
  const baseQ = g['trivia-champ'].questions[0]
  const q1 = {
    ...baseQ,
    id: 'q1',
    prompt: 'Question 1',
    timeLimit: q1Limit,
  }
  const qs = [q1]
  if (q2Limit !== null) {
    qs.push({
      ...baseQ,
      id: 'q2',
      prompt: 'Question 2',
      timeLimit: q2Limit,
    })
  }
  g['trivia-champ'].questions = qs
  g['trivia-champ'].rounds[0].questionIds = qs.map((q) => q.id)
  return g
}

export function buildDeck(concern, caseDef) {
  const witness = makeWitnessFixture(4, 'witness-timer')
  if (concern === 'leave-return') {
    const primary = caseDef.makeFixture('g-pri')
    return {
      ...deckWith([]),
      slides: [
        {
          id: 'p-slide',
          elements: [primary],
          children: [
            {
              id: 'c-slide',
              elements: [witness],
            },
          ],
        },
      ],
    }
  }
  if (concern === 'reset') {
    const primary = {
      ...caseDef.makeFixture('g-pri'),
      x: 20,
      y: 40,
      width: 440,
      height: 460,
    }
    const positionedWitness = {
      ...witness,
      x: 480,
      y: 40,
      width: 440,
      height: 460,
    }
    return {
      ...deckWith([]),
      slides: [
        {
          id: 'p-slide',
          elements: [primary, positionedWitness],
        },
      ],
    }
  }
  if (concern === 'isolation') {
    const f1 = {
      ...caseDef.f1('iso-1'),
      x: 20,
      y: 40,
      width: 440,
      height: 230,
    }
    const f2 = {
      ...caseDef.f2('iso-2'),
      x: 480,
      y: 40,
      width: 440,
      height: 230,
    }
    const positionedWitness = {
      ...witness,
      x: 20,
      y: 290,
      width: 440,
      height: 200,
    }
    return {
      ...deckWith([]),
      slides: [
        {
          id: 'p-slide',
          elements: [f1, f2, positionedWitness],
        },
      ],
    }
  }
  throw new Error(`Unknown concern: ${concern}`)
}
