export async function downloadArtifact(page, testInfo, menu, filename) {
  await page.getByRole('button', { name: 'File menu', exact: true }).click()
  const pending = page.waitForEvent('download', { timeout: 120000 })
  await page.getByRole('menuitem', { name: menu, exact: true }).click()
  const download = await pending
  const file = testInfo.outputPath(filename)
  await download.saveAs(file)
  expect(await download.failure()).toBeNull()
  return file
}

export async function freshPortable(browser, file, { initScripts = [], onPageCreated, evidence } = {}) {
  let context = null
  let page = null
  let handle = null
  const httpAttempts = []
  const socketAttempts = []
  const errors = []
  const pageErrors = []
  const frameSuccesses = []
  const frameFailures = []

  const recordArrival = (phase, err, meta) => {
    if (evidence?.recordError) {
      evidence.recordError(phase, err, meta)
    }
  }
  const getPrimaryError = () => evidence?.getPrimaryError?.() || null

  function rawDiagnostics() {
    let protocol = null
    try {
      const u = page?.url?.()
      if (u) protocol = new URL(u).protocol
    } catch {
      protocol = null
    }
    return {
      protocol,
      httpAttempts: [...httpAttempts],
      socketAttempts: [...socketAttempts],
      errors: [...errors],
      pageErrors: [...pageErrors],
      frameSuccesses: [...frameSuccesses],
      frameFailures: [...frameFailures],
    }
  }

  try {
    context = await browser.newContext({ reducedMotion: 'reduce' })
    await context.route(/^https?:\/\//i, (route) => {
      httpAttempts.push(route.request().url())
      return route.abort('blockedbyclient')
    })
    await context.routeWebSocket(/.*/, (socket) => {
      socketAttempts.push(socket.url())
      return socket.close({ code: 1008, reason: 'blockedbyclient' })
    })
    await context.addInitScript(() => {
      window.__releaseSockets = []
      const NativeSocket = window.WebSocket
      window.WebSocket = class extends NativeSocket {
        constructor(url, protocols) {
          window.__releaseSockets.push(String(url))
          super(url, protocols)
        }
      }
    })
    for (const script of initScripts) await context.addInitScript(script)
    page = await context.newPage()
    page.on('websocket', (socket) => socketAttempts.push(socket.url()))
    page.on('pageerror', (error) => {
      errors.push(error?.message || String(error))
      pageErrors.push(error)
      recordArrival('portable:pageerror', error)
    })

    async function inspectNetwork() {
      return await executePortableNetworkInspection({
        page,
        socketAttempts,
        httpAttempts,
        errors,
        frameSuccesses,
        frameFailures,
        recordArrival,
        getPrimaryError,
        rawDiagnostics,
        handle,
      })
    }

    handle = { page, context, file, diagnostics: rawDiagnostics(), inspectNetwork }
  } catch (setupError) {
    recordArrival('portable:setup', setupError)
    if (context) {
      try {
        await context.close()
      } catch (closeError) {
        if (!setupError.secondaryFailures) setupError.secondaryFailures = []
        setupError.secondaryFailures.push(closeError)
        recordArrival('portable:close', closeError)
      }
    }
    throw setupError
  }

  if (typeof onPageCreated === 'function') {
    await onPageCreated(handle)
    try {
      await handle.page.goto(pathToFileURL(file).href)
      await handle.page.waitForFunction(() => window.Reveal?.isReady())
      return handle
    } catch (navError) {
      handle.diagnostics = rawDiagnostics()
      recordArrival('portable:navigation', navError)
      throw navError
    }
  }

  try {
    await handle.page.goto(pathToFileURL(file).href)
    await handle.page.waitForFunction(() => window.Reveal?.isReady())
    return handle
  } catch (error) {
    handle.diagnostics = rawDiagnostics()
    recordArrival('portable:navigation', error)
    try {
      await context.close()
    } catch (closeError) {
      if (!error.secondaryFailures) error.secondaryFailures = []
      error.secondaryFailures.push(closeError)
      recordArrival('portable:close', closeError)
    }
    throw error
  }
}

export async function visitGeneratedSurfaces(
  { page, request, browser, id, testInfo, surfaces = ['present', 'share', 'file'] },
  visit
) {
  if (surfaces.includes('present')) {
    const { popup } = await launchPresentPopup(page)
    try {
      await popup.waitForFunction(() => window.Reveal?.isReady())
      await visit(popup, 'present', [])
    } finally {
      await popup.close()
    }
  }
  if (surfaces.includes('share')) {
    const share = await apiCreateShareLink(request, id)
    const sharePage = await page.context().newPage()
    try {
      await sharePage.goto(`/share/${share.token}`)
      await sharePage.waitForFunction(() => window.Reveal?.isReady())
      await visit(sharePage, 'share', [])
    } finally {
      await sharePage.close()
    }
  }
  if (surfaces.includes('file')) {
    const file = await downloadArtifact(page, testInfo, 'Export HTML', 'portable.html')
    const portable = await freshPortable(browser, file)
    try {
      const network = await portable.inspectNetwork()
      await visit(portable.page, 'file', [file], network)
      return await portable.inspectNetwork()
    } finally {
      await portable.context.close()
    }
  }
}

export async function selectGeneratedSlide(page, index) {
  await page.evaluate((index) => window.Reveal.slide(index), index)
  await expect(page.locator('.reveal .slides section.present').last()).toBeVisible()
}

export async function findFrame(page, selector) {
  let selected
  await expect.poll(async () => {
    for (const frame of page.frames()) {
      if (await frame.locator(selector).count()) {
        selected = frame
        return true
      }
    }
    return false
  }).toBe(true)
  return selected
}
